## 多账号隔离浏览器 - 开发文档

### 项目概述

开发一个支持多账号管理的浏览器工具，每个账号拥有独立的浏览器环境、指纹隔离、Cookie 存储，并完整支持 Chrome 扩展插件。

---

## 技术架构

### 技术栈选择

```
前端界面：Electron + React + Ant Design
浏览器内核：Puppeteer / Playwright（基于 Chromium）
后端逻辑：Node.js
数据存储：SQLite3（账号配置）+ 文件系统（Profile 数据）
代理支持：HTTP/SOCKS5 代理池
```

### 1. 浏览器管理器（browserManager.js）

```javascript
const puppeteer = require('puppeteer-extra');
const StealthPlugin = require('puppeteer-extra-plugin-stealth');
const path = require('path');
const fs = require('fs-extra');

puppeteer.use(StealthPlugin());

class BrowserManager {
  constructor() {
    this.browsers = new Map(); // accountId -> browser instance
  }

  /**
   * 启动独立浏览器实例
   * @param {Object} config - 账号配置
   */
  async launchBrowser(config) {
    const {
      accountId,
      profilePath,
      fingerprint,
      proxy,
      extensions,
      windowSize
    } = config;

    // 确保 Profile 目录存在
    await fs.ensureDir(profilePath);

    // 构建启动参数
    const args = [
      '--no-sandbox',
      '--disable-setuid-sandbox',
      '--disable-blink-features=AutomationControlled',
      `--window-size=${windowSize.width},${windowSize.height}`,
      '--disable-web-security', // 开发阶段使用，生产需谨慎
    ];

    // 加载扩展
    if (extensions && extensions.length > 0) {
      const extensionPaths = extensions
        .map(ext => path.resolve(__dirname, '../../extensions', ext))
        .join(',');
      args.push(
        `--disable-extensions-except=${extensionPaths}`,
        `--load-extension=${extensionPaths}`
      );
    }

    // 代理配置
    if (proxy && proxy.enabled) {
      args.push(`--proxy-server=${proxy.protocol}://${proxy.host}:${proxy.port}`);
    }

    // 启动浏览器
    const browser = await puppeteer.launch({
      headless: false,
      userDataDir: profilePath,
      args,
      defaultViewport: null,
      ignoreDefaultArgs: ['--enable-automation'],
    });

    // 注入反检测脚本
    const pages = await browser.pages();
    if (pages.length > 0) {
      await this.injectAntiDetect(pages[0], fingerprint);
    }

    // 监听新页面
    browser.on('targetcreated', async (target) => {
      if (target.type() === 'page') {
        const page = await target.page();
        if (page) await this.injectAntiDetect(page, fingerprint);
      }
    });

    this.browsers.set(accountId, browser);
    return browser;
  }

  /**
   * 注入反检测脚本
   */
  async injectAntiDetect(page, fingerprint) {
    await page.evaluateOnNewDocument((fp) => {
      // 移除 webdriver 特征
      Object.defineProperty(navigator, 'webdriver', {
        get: () => undefined
      });

      // 伪造 Chrome 对象
      window.navigator.chrome = {
        runtime: {},
        loadTimes: function() {},
        csi: function() {},
        app: {}
      };

      // 伪造 Permissions
      const originalQuery = window.navigator.permissions.query;
      window.navigator.permissions.query = (parameters) => (
        parameters.name === 'notifications' ?
          Promise.resolve({ state: Notification.permission }) :
          originalQuery(parameters)
      );

      // Canvas 指纹噪声
      const toBlob = HTMLCanvasElement.prototype.toBlob;
      const toDataURL = HTMLCanvasElement.prototype.toDataURL;
      const getImageData = CanvasRenderingContext2D.prototype.getImageData;
      
      const noisify = (data) => {
        for (let i = 0; i < data.length; i += 4) {
          data[i] = data[i] ^ fp.canvasNoise;
        }
        return data;
      };

      CanvasRenderingContext2D.prototype.getImageData = function() {
        const imageData = getImageData.apply(this, arguments);
        noisify(imageData.data);
        return imageData;
      };

      // WebGL 指纹伪造
      const getParameter = WebGLRenderingContext.prototype.getParameter;
      WebGLRenderingContext.prototype.getParameter = function(parameter) {
        if (parameter === 37445) return fp.webglVendor;
        if (parameter === 37446) return fp.webglRenderer;
        return getParameter.apply(this, arguments);
      };

      // 自定义 User-Agent
      Object.defineProperty(navigator, 'userAgent', {
        get: () => fp.userAgent
      });

      // 语言和时区
      Object.defineProperty(navigator, 'language', {
        get: () => fp.language
      });
      Object.defineProperty(navigator, 'languages', {
        get: () => fp.languages
      });

      // 屏幕分辨率
      Object.defineProperty(screen, 'width', { get: () => fp.screenWidth });
      Object.defineProperty(screen, 'height', { get: () => fp.screenHeight });

    }, fingerprint);

    // 代理认证（如需要）
    if (fingerprint.proxyAuth) {
      await page.authenticate({
        username: fingerprint.proxyAuth.username,
        password: fingerprint.proxyAuth.password
      });
    }
  }

  /**
   * 关闭浏览器实例
   */
  async closeBrowser(accountId) {
    const browser = this.browsers.get(accountId);
    if (browser) {
      await browser.close();
      this.browsers.delete(accountId);
    }
  }

  /**
   * 获取所有运行中的浏览器
   */
  getRunningBrowsers() {
    return Array.from(this.browsers.keys());
  }
}

module.exports = new BrowserManager();
```

### 2. 指纹生成引擎（fingerprintEngine.js）

```javascript
const crypto = require('crypto');

class FingerprintEngine {
  /**
   * 生成随机但一致的指纹配置
   * @param {string} seed - 种子（accountId）
   */
  generateFingerprint(seed) {
    const random = this.seededRandom(seed);

    return {
      // User-Agent
      userAgent: this.generateUserAgent(random),

      // 屏幕分辨率
      screenWidth: this.pickRandom(random, [1920, 1680, 1440, 2560]),
      screenHeight: this.pickRandom(random, [1080, 1050, 900, 1440]),

      // 语言
      language: this.pickRandom(random, ['zh-CN', 'en-US', 'ja-JP']),
      languages: this.pickRandom(random, [
        ['zh-CN', 'zh', 'en'],
        ['en-US', 'en'],
        ['ja-JP', 'ja', 'en']
      ]),

      // WebGL 指纹
      webglVendor: this.pickRandom(random, [
        'Google Inc. (NVIDIA)',
        'Google Inc. (Intel)',
        'Google Inc. (AMD)'
      ]),
      webglRenderer: this.pickRandom(random, [
        'ANGLE (NVIDIA GeForce RTX 3070)',
        'ANGLE (Intel UHD Graphics 630)',
        'ANGLE (AMD Radeon RX 5700)'
      ]),

      // Canvas 噪声值
      canvasNoise: Math.floor(random() * 10),

      // 时区
      timezone: this.pickRandom(random, [
        'Asia/Shanghai',
        'America/New_York',
        'Europe/London'
      ]),

      // 硬件并发数
      hardwareConcurrency: this.pickRandom(random, [4, 8, 12, 16]),

      // 设备内存（GB）
      deviceMemory: this.pickRandom(random, [4, 8, 16, 32]),
    };
  }

  /**
   * 生成 User-Agent
   */
  generateUserAgent(random) {
    const chromeVersions = ['120.0.6099.109', '119.0.6045.159', '121.0.6167.85'];
    const version = this.pickRandom(random, chromeVersions);
    
    const platforms = [
      `Windows NT 10.0; Win64; x64`,
      `Macintosh; Intel Mac OS X 10_15_7`,
      `X11; Linux x86_64`
    ];
    const platform = this.pickRandom(random, platforms);

    return `Mozilla/5.0 (${platform}) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/${version} Safari/537.36`;
  }

  /**
   * 基于种子的伪随机数生成器
   */
  seededRandom(seed) {
    const hash = crypto.createHash('sha256').update(seed).digest();
    let index = 0;
    
    return () => {
      if (index >= hash.length) index = 0;
      return hash[index++] / 255;
    };
  }

  pickRandom(random, array) {
    return array[Math.floor(random() * array.length)];
  }
}

module.exports = new FingerprintEngine();
```

### 3. Profile 管理器（profileManager.js）

```javascript
const path = require('path');
const fs = require('fs-extra');
const Database = require('./database');

class ProfileManager {
  constructor() {
    this.profilesDir = path.resolve(__dirname, '../../profiles');
    fs.ensureDirSync(this.profilesDir);
  }

  /**
   * 创建新账号 Profile
   */
  async createProfile(accountName, config = {}) {
    const accountId = this.generateAccountId();
    const profilePath = path.join(this.profilesDir, `account_${accountId}`);

    await fs.ensureDir(profilePath);

    // 保存到数据库
    await Database.insertAccount({
      id: accountId,
      name: accountName,
      profilePath,
      fingerprint: config.fingerprint || null,
      proxy: config.proxy || null,
      extensions: config.extensions || [],
      createdAt: Date.now()
    });

    return accountId;
  }

  /**
   * 删除账号
   */
  async deleteProfile(accountId) {
    const account = await Database.getAccount(accountId);
    if (account) {
      await fs.remove(account.profilePath);
      await Database.deleteAccount(accountId);
    }
  }

  /**
   * 获取账号配置
   */
  async getProfileConfig(accountId) {
    return await Database.getAccount(accountId);
  }

  /**
   * 更新账号配置
   */
  async updateProfile(accountId, updates) {
    await Database.updateAccount(accountId, updates);
  }

  generateAccountId() {
    return Date.now().toString(36) + Math.random().toString(36).substr(2);
  }
}

module.exports = new ProfileManager();
```

### 4. 数据库（database.js）

```javascript
const sqlite3 = require('sqlite3').verbose();
const path = require('path');
const fs = require('fs-extra');

class Database {
  constructor() {
    const dbDir = path.resolve(__dirname, '../../database');
    fs.ensureDirSync(dbDir);
    
    this.db = new sqlite3.Database(path.join(dbDir, 'accounts.db'));
    this.init();
  }

  init() {
    this.db.run(`
      CREATE TABLE IF NOT EXISTS accounts (
        id TEXT PRIMARY KEY,
        name TEXT NOT NULL,
        profilePath TEXT NOT NULL,
        fingerprint TEXT,
        proxy TEXT,
        extensions TEXT,
        createdAt INTEGER,
        lastUsed INTEGER
      )
    `);
  }

  async insertAccount(account) {
    return new Promise((resolve, reject) => {
      this.db.run(
        `INSERT INTO accounts (id, name, profilePath, fingerprint, proxy, extensions, createdAt)
         VALUES (?, ?, ?, ?, ?, ?, ?)`,
        [
          account.id,
          account.name,
          account.profilePath,
          JSON.stringify(account.fingerprint),
          JSON.stringify(account.proxy),
          JSON.stringify(account.extensions),
          account.createdAt
        ],
        (err) => err ? reject(err) : resolve()
      );
    });
  }

  async getAccount(id) {
    return new Promise((resolve, reject) => {
      this.db.get(
        `SELECT * FROM accounts WHERE id = ?`,
        [id],
        (err, row) => {
          if (err) return reject(err);
          if (!row) return resolve(null);
          
          resolve({
            ...row,
            fingerprint: JSON.parse(row.fingerprint),
            proxy: JSON.parse(row.proxy),
            extensions: JSON.parse(row.extensions)
          });
        }
      );
    });
  }

  async getAllAccounts() {
    return new Promise((resolve, reject) => {
      this.db.all(`SELECT * FROM accounts ORDER BY lastUsed DESC`, (err, rows) => {
        if (err) return reject(err);
        resolve(rows.map(row => ({
          ...row,
          fingerprint: JSON.parse(row.fingerprint),
          proxy: JSON.parse(row.proxy),
          extensions: JSON.parse(row.extensions)
        })));
      });
    });
  }

  async updateAccount(id, updates) {
    const fields = Object.keys(updates)
      .map(key => `${key} = ?`)
      .join(', ');
    
    const values = Object.values(updates).map(v => 
      typeof v === 'object' ? JSON.stringify(v) : v
    );

    return new Promise((resolve, reject) => {
      this.db.run(
        `UPDATE accounts SET ${fields} WHERE id = ?`,
        [...values, id],
        (err) => err ? reject(err) : resolve()
      );
    });
  }

  async deleteAccount(id) {
    return new Promise((resolve, reject) => {
      this.db.run(`DELETE FROM accounts WHERE id = ?`, [id], (err) => 
        err ? reject(err) : resolve()
      );
    });
  }
}

module.exports = new Database();
```

在 UI 中添加扩展管理功能：

```jsx
// renderer/components/ExtensionManager.jsx
import React, { useState } from 'react';
import { Upload, List, Button, message } from 'antd';
import { DeleteOutlined } from '@ant-design/icons';

export default function ExtensionManager({ accountId }) {
  const [extensions, setExtensions] = useState([]);

  const handleUpload = async (file) => {
    const formData = new FormData();
    formData.append('extension', file);
    formData.append('accountId', accountId);

    try {
      await window.electron.uploadExtension(formData);
      message.success('扩展上传成功');
      loadExtensions();
    } catch (err) {
      message.error('扩展上传失败');
    }
    
    return false; // 阻止默认上传行为
  };

  const loadExtensions = async () => {
    const list = await window.electron.getExtensions(accountId);
    setExtensions(list);
  };

  return (
    <div>
      <Upload
        beforeUpload={handleUpload}
        accept=".crx,.zip"
        showUploadList={false}
      >
        <Button>上传扩展</Button>
      </Upload>

      <List
        dataSource={extensions}
        renderItem={item => (
          <List.Item
            actions={[
              <Button 
                icon={<DeleteOutlined />} 
                onClick={() => deleteExtension(item.id)}
              />
            ]}
          >
            {item.name}
          </List.Item>
        )}
      />
    </div>
  );
}
```

```javascript
const { app, BrowserWindow, ipcMain } = require('electron');
const path = require('path');
const BrowserManager = require('./browserManager');
const ProfileManager = require('./profileManager');
const FingerprintEngine = require('./fingerprintEngine');

let mainWindow;

app.on('ready', () => {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false
    }
  });

  mainWindow.loadFile('dist/index.html');
});

// IPC 事件处理
ipcMain.handle('get-all-accounts', async () => {
  return await ProfileManager.getAllAccounts();
});

ipcMain.handle('create-account', async (event, name) => {
  const fingerprint = FingerprintEngine.generateFingerprint(name);
  
  return await ProfileManager.createProfile(name, {
    fingerprint,
    extensions: ['ublock_origin']
  });
});

ipcMain.handle('launch-browser', async (event, accountId) => {
  const config = await ProfileManager.getProfileConfig(accountId);
  
  await BrowserManager.launchBrowser({
    accountId: config.id,
    profilePath: config.profilePath,
    fingerprint: config.fingerprint,
    proxy: config.proxy,
    extensions: config.extensions,
    windowSize: { width: 1280, height: 800 }
  });
});

ipcMain.handle('close-browser', async (event, accountId) => {
  await BrowserManager.closeBrowser(accountId);
});
```
